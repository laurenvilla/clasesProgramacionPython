# 1.Investigar cómo se usa el operador módulo (modulus) en python, explicarlo con sus palabras y poner un ejemplo. (1 punto)
#un operador modulo se representa con un % y se usa para obtener el residuo de una division
# Por ejemplo, si el residuo al dividir entre 5 es 0, el número es múltiplo de 5.
number = 5
if number % 5 == 0:
    print(True)
else:
    print(False)
# En este caso el residuo es cero, por eso se imprime True.


#2. Crear una lista vacía y llenarla con los números pares del 2 al 50 (2,4,6,8,...,48,50) utilizando un for y un condicional para controlar que sólo los pares se agreguen.  (2 puntos)

lista = []
# Esta es la lista vacía donde se agregarán los números pares del 2 al 50.
for i in range(1, 51): # El límite superior del rango no se incluye.
    if i % 2 == 0: # Si el residuo al dividir entre 2 es cero, el número es par.
        lista.append(i) # Si se cumple la condición, agrega el número a la lista.
print(lista) # Imprime los números pares encontrados.


#3. Investigar para qué sirve el comando "break" en un for, explicarlo con sus palabras y hacer un ejemplo. (1 punto)
# break termina un ciclo antes de que se recorran todos sus valores.
# En este ejemplo se detiene el ciclo cuando encuentra el 12.
for n in range(12, 30):
    if n == 12:
        print("encontrado el 12")
        break
    print(n)

# También se puede detener el ciclo cuando se encuentra otro valor.
for x in range(70, 80):
    if x == 75:
        print(x)
        break


#4. Programar el método find de los strings a mano. (2 puntos) Sugerencia: utilizar break cuando encuentren lo que buscan
palabra = "banana"
buscar = "a"
posicion = -1 # -1 indica que todavía no se encontró la letra.


for indice, letra in enumerate(palabra):
    if letra == buscar:
        posicion = indice
        break # Como buscamos la primera aparición, detenemos el ciclo aquí.

print(posicion) # La primera "a" de "banana" está en el índice 1.
